Accepts Page as context */}} {{ $title := "" }} {{/* Override: termos de taxonomia (tags) aparecem como hashtag, ex.: #java */}} {{ if eq .Kind "term" }} {{ $title = printf "#%s" (.Data.Term | lower) }} {{ else if .LinkTitle }} {{ $title = .LinkTitle }} {{ else if .Title }} {{ $title = .Title }} {{ else if and .IsSection .File }} {{ $title = path.Base .File.Dir | humanize | title }} {{ else if and .IsPage .File }} {{ $title = .File.BaseFileName | humanize | title }} {{ else if .IsHome }} {{ $title = .Site.Title }} {{ end }} {{ return $title }}
